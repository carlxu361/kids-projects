const smooth = p => p * p * (3 - 2 * p);

const range = (min, max) => lerp(min, max, Math.random());

function shader(geometry, fragment, uniforms, options = {}, screen = false) {
  return new THREE.Mesh(geometry, new THREE.ShaderMaterial({
    uniforms,
    vertexShader: `
      varying vec2 vUv;
      void main() {
        vUv = uv;
        gl_Position = ${screen ? 'vec4(position.xy, 0., 1.)' : 'projectionMatrix * modelViewMatrix * vec4(position, 1.)'};
      }`,
    fragmentShader: fragment,
    ...options,
  }));
}

function sky() {
  const mesh = shader(new THREE.PlaneGeometry(2, 2), `
    varying vec2 vUv;
    uniform vec3 night, haze;
    uniform vec2 drift;
    void main() {
      vec2 p = vUv - .5;
      vec2 field = p - drift;
      float mist = exp(-4.5 * dot(field * vec2(.8, 1.5), field * vec2(.8, 1.5)));
      float horizon = exp(-12. * abs(field.y + .18));
      vec3 color = mix(night * .48, night, 1. - vUv.y);
      color += haze * (.31 * mist + .085 * horizon);
      color *= 1. - .6 * smoothstep(.13, .72, length(p));
      gl_FragColor = vec4(color, 1.);
      #include <colorspace_fragment>
    }`, {
    night: { value: new THREE.Color() }, haze: { value: new THREE.Color() },
    drift: { value: new THREE.Vector2() },
  }, { depthWrite: false, depthTest: false }, true);
  mesh.frustumCulled = false;
  mesh.renderOrder = -100;
  return mesh;
}

function halo(color, strength, screen = false) {
  const mesh = shader(new THREE.PlaneGeometry(2, 2), `
    varying vec2 vUv;
    uniform vec3 tint;
    uniform vec2 shape, center;
    uniform float gain;
    void main() {
      vec2 p = (vUv - center) * shape;
      float r2 = dot(p, p);
      float light = .65 * exp(-4.5 * r2) + .35 * exp(-16. * r2);
      gl_FragColor = vec4(tint, gain * light * (1. - smoothstep(.6, 1., r2)));
      #include <colorspace_fragment>
    }`, {
    tint: { value: new THREE.Color(color) }, gain: { value: strength },
    shape: { value: new THREE.Vector2(2, 2) }, center: { value: new THREE.Vector2(.5, .5) },
  }, {
    transparent: true, depthWrite: false, depthTest: !screen, blending: THREE.AdditiveBlending,
  }, screen);
  if (screen) {
    mesh.frustumCulled = false;
    mesh.renderOrder = 100;
  }
  return mesh;
}

function stars(capacity) {
  const positions = new THREE.BufferAttribute(new Float32Array(capacity * 3), 3);
  const colors = new THREE.BufferAttribute(new Float32Array(capacity * 3), 3);
  const sizes = new THREE.BufferAttribute(new Float32Array(capacity * 2), 2);
  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', positions);
  geometry.setAttribute('tint', colors);
  geometry.setAttribute('sizeAndAlpha', sizes);
  const material = new THREE.ShaderMaterial({
    uniforms: { time: { value: 0 }, pixels: { value: 1 } },
    vertexShader: `
      uniform float time, pixels;
      attribute vec3 tint;
      attribute vec2 sizeAndAlpha;
      varying vec3 vTint;
      varying float vAlpha;
      void main() {
        vec4 p = modelViewMatrix * vec4(position, 1.);
        gl_Position = projectionMatrix * p;
        gl_PointSize = clamp(sizeAndAlpha.x * pixels * 15. / max(2., -p.z), 1., 48.);
        vTint = tint;
        vAlpha = sizeAndAlpha.y * (.76 + .24 * sin(time * 1.65 + position.x * 7. + position.z * 3.))
          * smoothstep(.55, 2.2, -p.z) * (1. - smoothstep(65., 142., -p.z));
      }`,
    fragmentShader: `
      varying vec3 vTint;
      varying float vAlpha;
      void main() {
        vec2 p = 2. * gl_PointCoord - 1.;
        float r2 = dot(p, p);
        float light = (.32 * exp(-3. * r2) + .68 * exp(-19. * r2)) * (1. - smoothstep(.58, 1., r2));
        gl_FragColor = vec4(mix(vTint, vec3(1., .92, .8), exp(-30. * r2)), light * vAlpha);
        #include <colorspace_fragment>
      }`,
    transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
  });
  const points = new THREE.Points(geometry, material);
  points.frustumCulled = false;
  return {
    points,
    write(i, x, y, z, tint, size, opacity) {
      positions.setXYZ(i, x, y, z);
      colors.setXYZ(i, tint.r, tint.g, tint.b);
      sizes.setXY(i, size, opacity);
    },
    update(count, time, pixels) {
      geometry.setDrawRange(0, count);
      positions.needsUpdate = colors.needsUpdate = sizes.needsUpdate = true;
      this.clock(time, pixels);
    },
    clock(time, pixels) {
      material.uniforms.time.value = time;
      material.uniforms.pixels.value = pixels;
    },
  };
}

function makePlaque() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 512;
  const ctx = canvas.getContext('2d');
  const face = ctx.createRadialGradient(208, 162, 5, 258, 258, 240);
  face.addColorStop(0, '#4e2b52');
  face.addColorStop(.75, '#170f28');
  face.addColorStop(1, '#0c0919');
  ctx.beginPath();
  ctx.arc(256, 256, 234, 0, TAU);
  ctx.fillStyle = face;
  ctx.fill();
  for (const [radius, color, width] of [[229, '#bc8a56', 3], [211, '#ffd28b80', 1], [195, '#bd8b6a57', 1]]) {
    ctx.beginPath();
    ctx.arc(256, 256, radius, 0, TAU);
    ctx.lineWidth = width;
    ctx.strokeStyle = color;
    ctx.stroke();
  }
  for (let i = 0; i < 60; i++) {
    const angle = TAU * i / 60;
    const size = i % 5 ? 2 : 4;
    ctx.fillStyle = i % 5 ? '#e2b27e66' : '#ffdfae';
    ctx.fillRect(255 + 219 * Math.cos(angle), 255 + 219 * Math.sin(angle), size, size);
  }
  ctx.textAlign = 'center';
  ctx.font = 'bold 258px Georgia, serif';
  ctx.lineJoin = 'round';
  for (let offset = 11; offset > 0; offset--) {
    ctx.fillStyle = offset > 5 ? '#673324' : '#ba743a';
    ctx.fillText('12', 256 + offset * .42, 327 + offset);
  }
  const gold = ctx.createLinearGradient(0, 140, 0, 342);
  gold.addColorStop(0, '#fff7d9');
  gold.addColorStop(.47, '#ffd992');
  gold.addColorStop(.55, '#da8f47');
  gold.addColorStop(1, '#fff2c8');
  ctx.strokeStyle = '#ffeda6';
  ctx.lineWidth = 2.3;
  ctx.strokeText('12', 256, 327);
  ctx.fillStyle = gold;
  ctx.fillText('12', 256, 327);
  ctx.font = 'bold 22px monospace';
  ctx.fillStyle = '#f0c58e';
  ctx.fillText('LV.12 · 11.28', 256, 381);
  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.anisotropy = 4;
  return texture;
}

function metal(color, glow) {
  return new THREE.MeshStandardMaterial({
    color, emissive: glow, emissiveIntensity: .45, metalness: .64, roughness: .28,
  });
}

function portal(scene) {
  const group = new THREE.Group();
  scene.add(group);
  const gold = metal('#efaa52', '#af5024');
  const violet = metal('#644c83', '#4f295e');
  const rim = new THREE.Mesh(new THREE.TorusGeometry(1.42, .085, 24, 96), gold);
  const inner = new THREE.Mesh(new THREE.TorusGeometry(1.18, .05, 16, 72), violet);
  inner.position.z = -.12;
  const outline = new THREE.Mesh(new THREE.TorusGeometry(1.53, .013, 8, 96),
    new THREE.MeshBasicMaterial({ color: '#edbd76' }));
  outline.position.z = -.06;
  const glow = halo('#f09c63', .23);
  glow.scale.set(2.85, 2.85, 1);
  glow.position.z = -.7;
  const plaque = new THREE.Mesh(new THREE.PlaneGeometry(1.95, 1.95), new THREE.MeshBasicMaterial({
    map: makePlaque(), transparent: true, alphaTest: .4,
  }));
  plaque.position.z = -.23;
  const bevel = new THREE.Mesh(new THREE.CylinderGeometry(.89, .92, .11, 72), violet);
  bevel.rotation.x = Math.PI / 2;
  bevel.position.z = -.29;
  const orbit = new THREE.Group();
  const bead = new THREE.IcosahedronGeometry(.072, 1);
  for (let i = 0; i < 12; i++) {
    const star = new THREE.Mesh(bead, gold);
    const a = i * TAU / 12;
    star.position.set(Math.cos(a) * 1.66, Math.sin(a) * 1.66, .15 + Math.sin(a) * .09);
    orbit.add(star);
  }
  const key = new THREE.PointLight('#ffe2af', 24, 12);
  key.position.set(-2.5, 3.3, 4);
  const edge = new THREE.PointLight('#f08172', 13, 9);
  edge.position.set(2.7, -1.7, .8);
  const spot = new THREE.PointLight('#ffd18a', 0, 14);
  spot.position.set(.5, 1, 3.2);
  group.add(glow, rim, inner, outline, bevel, plaque, orbit, key, edge, spot);
  return { group, plaque, bevel, orbit, spot, glow };
}

function historySet(scene, z) {
  const grid = shader(new THREE.PlaneGeometry(44, 52), `
    varying vec2 vUv;
    uniform vec3 cyan, violet;
    void main() {
      vec2 cell = (vUv - .5) * vec2(44., 52.);
      vec2 edge = abs(fract(cell - .5) - .5);
      vec2 aa = max(fwidth(cell), vec2(.001));
      vec2 lines = 1. - smoothstep(aa * .25, aa * 1.2 + .016, edge);
      float line = max(lines.x, lines.y);
      float clear = 1. - smoothstep(.30, .85, max(abs(vUv.x - .5), abs(vUv.y - .5)) * 2.);
      float distanceFade = mix(.4, 1., smoothstep(.05, .8, vUv.y));
      gl_FragColor = vec4(mix(violet, cyan, vUv.x), clear * distanceFade * (.32 * line + .025));
      #include <colorspace_fragment>
    }`, { cyan: { value: new THREE.Color('#67c6b7') }, violet: { value: new THREE.Color('#9863d1') } },
  { transparent: true, depthWrite: false, blending: THREE.AdditiveBlending });
  grid.rotation.x = -Math.PI / 2;
  grid.position.set(0, -2.15, z - 11);
  scene.add(grid);
  const wire = new THREE.MeshBasicMaterial({
    color: '#b392e2', wireframe: true, transparent: true, opacity: .24, depthWrite: false,
  });
  const solids = Array.from({ length: 6 }, (_, i) => {
    const mesh = new THREE.Mesh(i % 2 ? new THREE.OctahedronGeometry(.62) : new THREE.IcosahedronGeometry(.7), wire);
    mesh.position.set((i - 2.5) * 2.85, range(-.65, 2.45), z - range(7, 14));
    scene.add(mesh);
    return mesh;
  });
  return solids;
}

function logSet(scene) {
  const orbits = [new THREE.Group(), new THREE.Group()];
  const stones = new THREE.IcosahedronGeometry(.105, 1);
  const colors = ['#ffbe75', '#c5a3ee'].map(tint => new THREE.MeshStandardMaterial({
    color: tint, emissive: tint, emissiveIntensity: .4, roughness: .38,
  }));
  orbits.forEach((orbit, i) => {
    orbit.position.set(0, .28, -35);
    orbit.rotation.x = i ? .66 : -.54;
    orbit.rotation.y = i ? -.37 : .42;
    const radius = i ? 2.65 : 3.5;
    const count = i ? 9 : 10;
    const ellipse = new THREE.EllipseCurve(0, 0, radius, radius * .53);
    const geometry = new THREE.BufferGeometry().setFromPoints(ellipse.getPoints(72));
    orbit.add(new THREE.LineLoop(geometry, new THREE.LineBasicMaterial({
      color: i ? '#9674bd' : '#d49a68', opacity: .2, transparent: true, depthWrite: false,
    })));
    for (let n = 0; n < count; n++) {
      const gem = new THREE.Mesh(stones, colors[(n + i) % 2]);
      const p = ellipse.getPoint(n / count);
      gem.position.set(p.x, p.y, .04);
      orbit.add(gem);
    }
    scene.add(orbit);
  });
  return orbits;
}

function scenery(scene) {
  const layers = [280, 215, 150].map((count, depth) => {
    const layer = stars(count);
    const tint = new THREE.Color(['#aa9bdc', '#e9b993', '#ddaece'][depth]);
    for (let i = 0; i < count; i++) {
      layer.write(i, range(-30, 30), range(-15, 15), range(-185, 4 - depth * 2),
        tint, range(2.3, 5.8) + depth * 1.4, range(.27, .76));
    }
    layer.update(count, 0, 1);
    scene.add(layer.points);
    return layer;
  });
  const entry = portal(scene);
  const solids = historySet(scene, views.history[0]);
  const orbits = logSet(scene);
  const pianoCurtains = Array.from({ length: 3 }, (_, i) => {
    const curtain = halo(i === 1 ? '#d7a17b' : '#b4808b', .11);
    curtain.position.set((i - 1) * 3.4, .5, -50 - i);
    curtain.scale.set(.92, 7, 1);
    scene.add(curtain);
    return curtain;
  });
  const nebula = stars(174);
  const violet = new THREE.Color('#a679eb');
  for (let i = 0; i < 174; i++) {
    const p = i / 174;
    const angle = p * TAU * 2.8 + (i % 2) * Math.PI;
    const radius = lerp(.8, 5.7, p);
    nebula.write(i, Math.cos(angle) * radius + range(-.45, .45),
      Math.sin(angle) * radius * .52 + range(-.25, .25), range(-1.8, 1.8),
      violet, range(3, 10), range(.22, .64));
  }
  nebula.points.position.set(0, .25, -75);
  nebula.points.rotation.y = -.28;
  nebula.update(174, 0, 1);
  scene.add(nebula.points);
  const lanterns = Array.from({ length: 16 }, (_, i) => {
    const mote = halo(i % 3 ? '#e6a864' : '#ed8c76', range(.09, .25));
    mote.position.set(range(-11, 11), range(-3, 4), range(-90, -83));
    mote.scale.setScalar(range(.35, 1.55));
    scene.add(mote);
    return { mote, height: mote.position.y };
  });
  const display = halo('#efab6a', .17);
  display.scale.set(8, 4, 1);
  display.position.set(0, -.9, -98);
  scene.add(display);
  const finale = stars(155);
  const plum = new THREE.Color('#eab4c1');
  for (let i = 0; i < 155; i++) {
    finale.write(i, range(-20, 20), range(-9, 8), range(-150, -119), plum, range(5, 13), range(.43, .84));
  }
  finale.update(155, 0, 1);
  scene.add(finale.points);
  return {
    entry,
    tick(t, pixels, party) {
      layers.forEach((layer, i) => {
        layer.points.position.y = Math.sin(t * (.13 + i * .045) + i) * (.2 + i * .08);
        layer.clock(t, pixels);
      });
      entry.orbit.rotation.z = t * .105;
      entry.spot.intensity = 32 * party;
      entry.glow.material.uniforms.gain.value = .23 + .24 * party;
      solids.forEach((m, i) => m.rotation.set(t * .075 + i, t * .11 + i, .2 * i));
      orbits.forEach((m, i) => { m.rotation.z = t * (i ? -.17 : .115); });
      pianoCurtains.forEach((m, i) => { m.position.x = (i - 1) * 3.4 + Math.sin(t * .12 + i) * .25; });
      nebula.points.rotation.z = t * .065;
      nebula.clock(t, pixels);
      lanterns.forEach(({ mote, height }, i) => { mote.position.y = height + .43 * Math.sin(t * .24 + i * 1.5); });
      finale.points.position.y = .45 * Math.sin(t * .24);
      finale.clock(t, pixels);
    },
  };
}

function bursts(scene) {
  const sparkLimit = 384;
  const bed = stars(sparkLimit);
  scene.add(bed.points);
  const sparks = [];
  const rockets = [];
  const gift = new THREE.Group();
  scene.add(gift);
  const tones = ['#ffce7a', '#ff9d86', '#c7a8ff'].map(hex => new THREE.Color(hex));
  const gem = new THREE.OctahedronGeometry(.085);
  const materials = tones.map(tone => new THREE.MeshBasicMaterial({ color: tone, transparent: true }));
  const gems = Array.from({ length: 72 }, (_, i) => {
    const mesh = new THREE.Mesh(gem, materials[i % materials.length]);
    gift.add(mesh);
    return { mesh, velocity: new THREE.Vector3(), spin: new THREE.Vector3() };
  });
  const ring = new THREE.Mesh(new THREE.TorusGeometry(1, .025, 8, 72),
    new THREE.MeshBasicMaterial({ color: '#ffe0a7', transparent: true, depthWrite: false }));
  gift.add(ring);
  gift.visible = false;
  let giftTime = 0;

  function velocity(out, speed) {
    const y = range(-1, 1);
    const angle = range(0, TAU);
    const radius = Math.sqrt(1 - y * y) * speed;
    return out.set(Math.cos(angle) * radius, y * speed, Math.sin(angle) * radius);
  }

  return {
    clear() { sparks.length = rockets.length = 0; gift.visible = false; },
    launch(x, y, z, delay = 0) {
      rockets.push({ x, y: y - 3.7, z, vy: range(8.8, 10.7), age: -delay, tone: tones[rockets.length % tones.length] });
    },
    gift(x, y, z) {
      gift.visible = true;
      gift.position.set(x, y, z);
      giftTime = 0;
      for (const { mesh, velocity: v, spin } of gems) {
        mesh.position.set(0, 0, 0);
        mesh.rotation.set(range(0, TAU), range(0, TAU), range(0, TAU));
        mesh.scale.setScalar(range(.65, 1.55));
        velocity(v, range(1.5, 8.1));
        v.y += 1.7;
        spin.set(range(-7, 7), range(-7, 7), range(-7, 7));
      }
    },
    tick(dt, t, pixels) {
      let count = 0;
      for (let i = rockets.length - 1; i >= 0; i--) {
        const r = rockets[i];
        r.age += dt;
        if (r.age < 0) continue;
        r.y += r.vy * dt;
        r.vy -= 8 * dt;
        if (r.age < .53) {
          for (let j = 0; j < 7 && count < sparkLimit; j++) {
            bed.write(count++, r.x, r.y - j * .11, r.z, r.tone, 8 - j * .6, 1 - j * .13);
          }
          continue;
        }
        rockets.splice(i, 1);
        for (let j = 0; j < 48; j++) {
          sparks.push({ x: r.x, y: r.y, z: r.z, v: velocity(new THREE.Vector3(), range(1.5, 5.3)),
            age: 0, life: range(.72, 1.65), tone: r.tone, size: range(6, 12) });
        }
      }
      for (let i = sparks.length - 1; i >= 0; i--) {
        const p = sparks[i];
        p.age += dt;
        if (p.age >= p.life) { sparks.splice(i, 1); continue; }
        p.v.multiplyScalar(Math.exp(-1.15 * dt));
        p.v.y -= 1.9 * dt;
        p.x += dt * p.v.x;
        p.y += dt * p.v.y;
        p.z += dt * p.v.z;
        if (count < sparkLimit) bed.write(count++, p.x, p.y, p.z, p.tone, p.size, (1 - p.age / p.life) ** 1.5);
      }
      bed.update(count, t, pixels);
      if (!gift.visible) return;
      giftTime += dt;
      gift.visible = giftTime < 1.7;
      const drag = Math.exp(-1.55 * dt);
      materials.forEach(material => { material.opacity = 1 - smooth(clamp((giftTime - .55) / 1.15, 0, 1)); });
      gems.forEach(({ mesh, velocity: v, spin }) => {
        v.multiplyScalar(drag);
        v.y -= 5.7 * dt;
        mesh.position.addScaledVector(v, dt);
        mesh.rotation.x += dt * spin.x;
        mesh.rotation.y += dt * spin.y;
        mesh.rotation.z += dt * spin.z;
      });
      ring.scale.setScalar(.12 + 4.4 * (1 - Math.exp(-2.7 * giftTime)));
      ring.material.opacity = .72 * Math.max(0, 1 - giftTime / .72);
    },
  };
}

export function mountWorld3D(canvas, { initialChapter, fireworksAllowed, onReady, onFallback }) {
  const gl = canvas.getContext('webgl2', { alpha: false, antialias: true, powerPreference: 'high-performance' });
  if (!gl || gl.isContextLost()) throw Error('WebGL2 unavailable');
  const renderer = new THREE.WebGLRenderer({ canvas, context: gl, antialias: true, alpha: false, powerPreference: 'high-performance' });
  renderer.debug.onShaderError = () => { throw Error('Cinema shader could not be linked'); };
  const scene = new THREE.Scene();
  let art, background, effects, light;
  try {
    background = sky();
    light = halo('#ffbc83', 0, true);
    scene.add(background, light, new THREE.HemisphereLight('#ffdcb7', '#50476e', 1.45));
    art = scenery(scene);
    effects = bursts(scene);
  } catch (error) {
    disposeScene();
    throw error;
  }
  const camera = new THREE.PerspectiveCamera(46, 1, .055, 180);
  const position = new THREE.Vector3();
  const target = new THREE.Vector3();
  const destination = new THREE.Vector3();
  const pointer = new THREE.Vector2();
  const parallax = new THREE.Vector2();
  const fine = matchMedia('(pointer: fine)');
  const night = new THREE.Color();
  const haze = new THREE.Color();
  const nextNight = new THREE.Color();
  const nextHaze = new THREE.Color();
  let act = viewKey(initialChapter);
  let previewTo;
  let flight;
  let frame = 0;
  let last = 0;
  let time = 0;
  let launchAt = 2.2;
  let party = 0;
  let pixels = 1;
  let width = 0, height = 0, rawDpr = 0;
  let quality = 0, ema = 16, slow = 0, fast = 0;
  let paused = false, contextLost = false, destroyed = false, ready = false;
  let enabled = Boolean(fireworksAllowed);

  function viewKey(key) { return Object.hasOwn(views, key) ? key : 'intro'; }

  function place(key, out) { return out.set(0, 0, views[viewKey(key)][0]); }

  function resize() {
    if (destroyed || contextLost || paused) return;
    const w = Math.max(1, innerWidth);
    const h = Math.max(1, innerHeight);
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const budget = Math.min(1, Math.sqrt((2560 * 1600) / Math.max(1, w * h * dpr * dpr)));
    const resolution = dpr * budget * qualityScales[quality];
    if (w === width && h === height && dpr === rawDpr && resolution === pixels) return;
    width = w;
    height = h;
    rawDpr = dpr;
    pixels = resolution;
    renderer.setPixelRatio(pixels);
    renderer.setSize(w, h, false);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    const compact = w <= 680;
    const ringScale = compact ? clamp(camera.aspect * 1.05, .37, .79) : 1;
    art.entry.group.position.set(compact ? 0 : Math.min(2.05, camera.aspect * 1.15), compact ? .65 : .05, 2.2);
    art.entry.group.scale.setScalar(ringScale);
  }

  function adapt(ms) {
    ema += .08 * (ms - ema);
    slow = ema > 24 ? slow + ms : 0;
    fast = ema < 17 ? fast + ms : 0;
    const next = slow > 480 ? Math.min(quality + 1, qualityScales.length - 1) : fast > 5000 ? Math.max(quality - 1, 0) : quality;
    if (next !== quality) {
      quality = next;
      slow = fast = 0;
      resize();
    }
  }

  function setAct(key) {
    act = viewKey(key);
    launchAt = time + 1.5;
  }

  function snapTo(key) {
    cancel();
    setAct(key);
    place(act, position);
    night.set(views[act][1]);
    haze.set(views[act][2]);
  }

  function preview(from, to) {
    if (flight) return;
    previewTo = { from, to };
  }

  function previewPosition() {
    place(act, target);
    if (!previewTo) return;
    if (previewTo.from === 'intro' && previewTo.to === 'history') {
      const anchor = art.entry.group.position;
      target.set(anchor.x * .78, anchor.y * .78, 5.85);
    } else {
      target.z -= .46;
      target.x = previewTo.from === 'piano' && previewTo.to === 'cake' ? 0 : .18;
    }
  }

  function schedule() {
    if (!frame && !paused && !contextLost && !destroyed && !document.hidden) frame = requestAnimationFrame(tick);
  }

  function suspendFrame() {
    cancelAnimationFrame(frame);
    frame = 0;
    last = 0;
    ema = 16;
    slow = fast = 0;
  }

  function cancel() {
    flight = previewTo = null;
  }

  function travel({ from, to, onCommit, onDone }) {
    previewTo = null;
    const entering = from === 'intro' && to === 'history';
    const daybreak = from === 'piano' && to === 'cake';
    flight = {
      to, entering, daybreak, onCommit, onDone,
      duration: entering ? .78 : daybreak ? .74 : .52,
      elapsed: 0, rate: 1, committed: false, start: position.clone(),
      startNight: night.clone(), startHaze: haze.clone(),
    };
    if (from === 'cake' && paused) {
      // The cake must dispose its renderer before the world can draw another frame.
      flight.committed = true;
      onCommit();
    }
    schedule();
  }

  function paint(glow, progress, daybreak) {
    const breathing = flight?.entering ? Math.abs(2 * progress - 1) : 1;
    camera.position.set(position.x + Math.sin(time * .21) * .14 * breathing,
      position.y + Math.cos(time * .17) * .09 * breathing, position.z);
    camera.rotation.set(parallax.y * degToRad(2.2) * breathing,
      -parallax.x * degToRad(3.2) * breathing, 0, 'YXZ');
    camera.fov = lerp(46, 55, smooth(clamp((-position.z - 95) / 11, 0, 1)));
    camera.updateProjectionMatrix();
    background.material.uniforms.night.value.copy(night);
    background.material.uniforms.haze.value.copy(haze);
    background.material.uniforms.drift.value.set(parallax.x * .035, parallax.y * .025);
    light.visible = glow > .001;
    light.material.uniforms.gain.value = glow;
    light.material.uniforms.center.value.y = daybreak ? lerp(.54, .44, progress) : .5;
    light.material.uniforms.shape.value.set(
      Math.max(1.7, camera.aspect * (daybreak ? lerp(1.34, 2.6, progress) : 1.5)),
      daybreak ? lerp(1.52, 2.65, progress) : 1.72,
    );
    renderer.render(scene, camera);
    if (gl.isContextLost()) throw Error('WebGL context lost');
  }

  function animateFlight(move, p) {
    place(move.to, destination);
    if (move.entering) {
      const gate = art.entry.group.position;
      if (p < .46) {
        const q = smooth(p / .46);
        position.set(lerp(move.start.x, gate.x, q), lerp(move.start.y, gate.y, q),
          lerp(move.start.z, gate.z, p / .46));
      } else {
        const q = ease((p - .46) / .54);
        position.set(lerp(gate.x, destination.x, smooth(q)), lerp(gate.y, destination.y, smooth(q)),
          lerp(gate.z, destination.z, q));
      }
    } else {
      position.lerpVectors(move.start, destination, ease(p));
    }
    nextNight.set(views[viewKey(move.to)][1]);
    nextHaze.set(views[viewKey(move.to)][2]);
    night.lerpColors(move.startNight, nextNight, smooth(p));
    haze.lerpColors(move.startHaze, nextHaze, smooth(p));
  }

  function tick(now) {
    frame = 0;
    if (paused || contextLost || destroyed) return;
    const ms = last ? now - last : 16.7;
    last = now;
    const dt = Math.min(ms / 1000, .065);
    time += dt;
    if (ready) adapt(Math.min(ms, 100));
    const damping = 1 - Math.exp(-5 * dt);
    parallax.lerp(pointer, damping);
    party *= Math.exp(-dt * .7);
    const move = flight;
    let p = 0;
    let glow = 0;
    if (move) {
      move.elapsed += dt * move.rate;
      p = clamp(move.elapsed / move.duration, 0, 1);
      animateFlight(move, p);
      glow = (p === 1 ? 0 : Math.sin(Math.PI * p)) * (move.daybreak ? .48 : move.entering ? .3 : .065);
    } else {
      previewPosition();
      position.lerp(target, damping);
      if (previewTo?.from === 'piano' && previewTo.to === 'cake') glow = .14 * clamp((views.piano[0] - position.z) / .46, 0, 1);
    }
    // Clear the center just before the lens reaches the portal; the torus still passes around it.
    const passage = move?.entering ? 1 - smooth(clamp((p - .24) / .17, 0, 1)) : 1;
    art.entry.plaque.scale.setScalar(Math.max(.001, passage));
    art.entry.bevel.scale.copy(art.entry.plaque.scale);
    if (enabled && !move && climaxActs.has(act) && time >= launchAt) {
      effects.launch(range(-5.7, 5.7), range(1.6, 3.1), position.z - range(8, 14));
      launchAt = time + range(3.2, 5.4);
    }
    art.tick(time, pixels, party);
    effects.tick(dt, time, pixels);
    try {
      paint(glow, p, move?.daybreak || previewTo?.to === 'cake');
    } catch {
      lost();
      return;
    }

    if (!ready) {
      ready = true;
      onReady?.();
    }
    if (move && flight === move && !move.committed && p >= (move.daybreak ? .48 : .46)) {
      move.committed = true;
      if (move.to === 'cake') {
        // Leave an uncolored cake backdrop in the parked drawing buffer.
        animateFlight(move, 1);
        try { paint(0, 1, false); } catch { lost(); return; }
      }
      move.onCommit();
    }
    if (move && flight === move && (p === 1 || (move.committed && paused))) {
      flight = null;
      move.onDone();
    }
    schedule();
  }

  function point(event) {
    if (!fine.matches || event.pointerType === 'touch') return;
    pointer.set(clamp(event.clientX / Math.max(1, width) * 2 - 1, -1, 1),
      clamp(1 - event.clientY / Math.max(1, height) * 2, -1, 1));
  }

  function resetPointer() { pointer.set(0, 0); }

  function lost(event) {
    event?.preventDefault();
    if (contextLost || destroyed) return;
    contextLost = true;
    suspendFrame();
    onFallback?.();
  }

  function restored() {
    if (destroyed) return;
    contextLost = false;
    schedule();
  }

  function disposeScene() {
    const resources = new Set();
    scene.traverse(node => {
      if (node.geometry) resources.add(node.geometry);
      if (node.material?.map) resources.add(node.material.map);
      if (node.material) resources.add(node.material);
    });
    resources.forEach(resource => resource.dispose());
    renderer.dispose();
  }

  const observer = new ResizeObserver(resize);
  observer.observe(document.documentElement);
  window.addEventListener('resize', resize);
  window.addEventListener('pointermove', point, { passive: true });
  document.documentElement.addEventListener('pointerleave', resetPointer);
  canvas.addEventListener('webglcontextlost', lost);
  canvas.addEventListener('webglcontextrestored', restored);
  snapTo(act);
  resize();
  schedule();

  return {
    snapTo, preview, travel, cancel, setAct,
    accelerate() { if (flight) flight.rate = 3.5; },
    setFireworksEnabled(value) {
      enabled = Boolean(value);
      if (!enabled) effects.clear();
    },
    triggerPartyLight() {
      party = 1;
      if (enabled) for (const [index, x] of [-3.4, 3.8, -.6].entries()) {
        effects.launch(x, 1.65 + index * .38, position.z - 7.5 - index * 1.7, index * .16);
      }
    },
    triggerGiftBurst() { if (enabled) effects.gift(0, .1, views.shop[0] - 5.1); },
    pause() { paused = true; suspendFrame(); },
    resume() { paused = false; resize(); schedule(); },
    isContextLost() { return contextLost || gl.isContextLost(); },
    destroy() {
      if (destroyed) return;
      destroyed = true;
      suspendFrame();
      cancel();
      observer.disconnect();
      window.removeEventListener('resize', resize);
      window.removeEventListener('pointermove', point);
      document.documentElement.removeEventListener('pointerleave', resetPointer);
      canvas.removeEventListener('webglcontextlost', lost);
      canvas.removeEventListener('webglcontextrestored', restored);
      disposeScene();
    },
  };
}
